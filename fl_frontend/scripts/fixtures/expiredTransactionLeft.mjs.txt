// A db file that leaves a transaction to MongoDB's lifetime limit, run as a child by
// `fl_frontend/src/core/expiredTransactions.db.test.ts`; the teardown the helper registered fails it.
import assert from "node:assert/strict";
import { it } from "node:test";

import { startJudgedReplicaSet } from "@/core/expiredTransactions.ts";

const { mongod } = await startJudgedReplicaSet();

// In the container's own shell, so no client of this file holds the transaction: one left open past a
// one-second lifetime, waited on until the server's periodic pass has aborted it.
const LEAVE_ONE_TO_EXPIRE = `
db.adminCommand({ setParameter: 1, transactionLifetimeLimitSeconds: 1 });
const session = db.getMongo().startSession();
session.startTransaction();
session.getDatabase("probe").rows.insertOne({ left: true });
const killed = () => Number(db.serverStatus().metrics.abortExpiredTransactions.successfulKills);
const deadline = Date.now() + 30000;
while (killed() < 1 && Date.now() < deadline) sleep(200);
print(killed());
`;

it("leaves a transaction to expire", async () => {
  const { output, exitCode } = await mongod.exec(["mongosh", "--quiet", "--eval", LEAVE_ONE_TO_EXPIRE]);

  assert.equal(exitCode, 0, output);
  assert.equal(output.trim(), "1", "the server never aborted the transaction left open");
});
