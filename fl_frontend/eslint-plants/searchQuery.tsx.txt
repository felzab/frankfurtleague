declare const searchParams: URLSearchParams;
declare const search: string;
declare const pathname: string;
declare const typed: string;
declare const item: { q: string; a: string };
declare const props: { searchParams: Promise<Record<string, string | undefined>> };
declare const router: { replace: (to: string) => void };

export const READ = searchParams.get("q"); // expect: search-in-url
export const READ_ALL = searchParams.getAll(`q`); // expect: search-in-url
export const BUILT = new URLSearchParams({ q: typed }); // expect: search-in-url
export const LINK = `/teams?q=${typed}`; // expect: search-in-url
export const LOADED_FIELD = import("@heroui/react/search-field"); // expect: search-field

export function write(): void {
  const params = new URLSearchParams(search);
  params.set("q", typed); // expect: search-in-url
  params.append("q", typed); // expect: search-in-url
  router.replace(`${pathname}?saison_id=1&q=${typed}`); // expect: search-in-url
}

export async function readOnTheServer(): Promise<unknown[]> {
  const params = await props.searchParams;
  const { q } = await props.searchParams; // expect: search-in-url
  return [params.q, params["q"], (await props.searchParams).q, q]; // expect: search-in-url expect: search-in-url expect: search-in-url
}

      <input type="search" aria-label="Suchen" /> {/* expect: search-field */}
      <input role="searchbox" aria-label="Suchen" /> {/* expect: search-field */}
    </>
  );
}
