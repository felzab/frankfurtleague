export function write(): void {
  const params = new URLSearchParams(search);
  params.set("q", typed); // expect: search-in-url
  params.append("q", typed); // expect: search-in-url
  router.replace(`${pathname}?saison_id=1&q=${typed}`); // expect: search-in-url
}

export async function readOnTheServer(): Promise<unknown[]> {
  const params = await props.searchParams;
  const { q } = await props.searchParams; // expect: search-in-url
  return [params.q, params["q"], (await props.searchParams).q, q]; // expect: search-in-url expect: search-in-url expect: search-in-url
}

// The parameters held under any name: through an assertion, a fallback, an alias and a rest element.
export async function readABinding(): Promise<unknown[]> {
  const taken = await props.searchParams;
  const kept = (await searchParamsPromise) ?? {};
  const alias = taken as Record<string, string | undefined>;
  const { q: query } = alias; // expect: search-in-url
  const { saison_id: saison, ...rest } = kept;
  const { q: fallen } = (await props.searchParams) ?? {}; // expect: search-in-url
  return [taken.q, kept[`q`], rest.q, query, saison, fallen]; // expect: search-in-url expect: search-in-url expect: search-in-url
}

// A client page's parameters, through React's `use`.
export function ReadInTheBrowser() {
  const used = use(props.searchParams);
  return [used.q, use(props.searchParams).q]; // expect: search-in-url expect: search-in-url
}

// Free: a binding of anything but the page's parameters, and one destructured key's value.
export async function readOtherBindings(): Promise<unknown[]> {
  const antwort = await readFaq();
  const { saison_id: saison } = await props.searchParams;
  return [antwort.q, (saison as unknown as { q: string }).q];
}
