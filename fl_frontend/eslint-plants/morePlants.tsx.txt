declare const appRouter: { push: (to: string) => void };
declare const shell: { router: { replace: (to: string) => void } };
declare const response: { redirect: (to: string) => void };
declare const permanentRedirect: (to: string) => never;

export function plantedNavigations(): void {
  appRouter.push("teams"); // expect: relative-nav
  shell.router.replace(`teams/${String(appRouter)}`); // expect: relative-nav
  response.redirect("teams"); // expect: relative-nav
  permanentRedirect("teams"); // expect: relative-nav
}

// A target keeps the text its leftmost operand opens on through each wrapper.
export function plantedWrappedTargets(): void {

  appRouter.push(`teams/${String(appRouter)}` + "/kader" + String(appRouter)); // expect: relative-nav

  appRouter.push(("teams/" + String(appRouter)) satisfies string); // expect: relative-nav
  appRouter.push(("teams" as string | null)!); // expect: relative-nav
  appRouter.push("teams/".concat(String(appRouter))); // expect: relative-nav
}
